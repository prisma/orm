#!/bin/zsh
# usage: measure.sh <label> [probe-file] [demo-contract-patch-script]   (run from repo root)
set -u
export PATH=/Users/wmadden/.local/share/mise/installs/node/24.13.0/bin:$PATH
setopt null_glob
ROOT=$(pwd)
LABEL=$1
PROBE=${2:-}
PATCH=${3:-}
OUT=$ROOT/wip/spike/out/$LABEL
mkdir -p $OUT
PKG=$ROOT/packages/3-extensions/sql-orm-client
DEMO=$ROOT/examples/prisma-8-demo

pnpm build > $OUT/build.log 2>&1 || { echo "BUILD FAILED"; tail -40 $OUT/build.log; exit 1; }

cat > $PKG/tsconfig.noscopes.json <<EOT
{ "extends": "./tsconfig.json", "exclude": ["dist","dist-tsc","dist-tsc-prod","node_modules","test/scopes*"] }
EOT
for run in 1 2; do
(cd $PKG && pnpm typecheck -p tsconfig.noscopes.json --extendedDiagnostics > $OUT/pkg-noscopes-$run.txt 2>&1; echo "exit $?" >> $OUT/pkg-noscopes-$run.txt)
(cd $PKG && pnpm typecheck --extendedDiagnostics > $OUT/pkg-withscopes-$run.txt 2>&1; echo "exit $?" >> $OUT/pkg-withscopes-$run.txt)
done
rm $PKG/tsconfig.noscopes.json

rm -f $DEMO/test/zz-*.test-d.ts
for run in 1 2; do
(cd $DEMO && pnpm typecheck --extendedDiagnostics > $OUT/demo-noscopes-$run.txt 2>&1; echo "exit $?" >> $OUT/demo-noscopes-$run.txt)
done
if [ -f $ROOT/wip/spike/notany-probe.ts ]; then
cp $ROOT/wip/spike/notany-probe.ts $DEMO/test/zz-notany.test-d.ts
(cd $DEMO && pnpm typecheck > $OUT/demo-notany.txt 2>&1; echo "exit $?" >> $OUT/demo-notany.txt)
rm -f $DEMO/test/zz-notany.test-d.ts
fi
if [ -n "$PROBE" ]; then
  if [ -n "$PATCH" ]; then
    cp $DEMO/src/prisma/contract.d.ts $OUT/contract.d.ts.orig
    node $PATCH $DEMO/src/prisma/contract.d.ts || exit 1
  fi
  cp $PROBE $DEMO/test/zz-probe.test-d.ts
  for run in 1 2; do
  (cd $DEMO && pnpm typecheck --extendedDiagnostics > $OUT/demo-withscopes-$run.txt 2>&1; echo "exit $?" >> $OUT/demo-withscopes-$run.txt)
  done
  rm -f $DEMO/test/zz-probe.test-d.ts
  if [ -n "$PATCH" ]; then cp $OUT/contract.d.ts.orig $DEMO/src/prisma/contract.d.ts; fi
fi
for f in $OUT/pkg-*.txt $OUT/demo-*.txt; do echo "== $f"; grep -E '^(Instantiations|Types|exit)|error TS' $f | head -12; done
